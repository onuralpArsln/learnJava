$$$$ Table Creation 

CREATE TABLE Movies (
    ID INT IDENTITY(1,1),           -- auto-incrementing movie ID
    title VARCHAR(50) NOT NULL,
    year INT,
    length DECIMAL(8,2),            -- 6 digits before + 2 after decimal
    studioName VARCHAR(30),
    producerID INT,
    PRIMARY KEY (ID, title)
);
